1. Aplikacja z Listą wyboru (ComboBox)

Aplikacja z ComboBox, która pozwala użytkownikowi wybrać element z listy.

<Window x:Class="ComboBoxExample.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        Title="ComboBox Example" Height="200" Width="300">
    <Grid>
        <ComboBox Name="comboBox" HorizontalAlignment="Center" VerticalAlignment="Center" Width="200">
            <ComboBoxItem>Option 1</ComboBoxItem>
            <ComboBoxItem>Option 2</ComboBoxItem>
            <ComboBoxItem>Option 3</ComboBoxItem>
        </ComboBox>
        <Button Content="Show Selected" VerticalAlignment="Bottom" HorizontalAlignment="Center" Margin="10" Click="Button_Click"/>
    </Grid>
</Window>

W C#: ---------------------------------------------------------------------------------------------------------------------------------------------------

private void Button_Click(object sender, RoutedEventArgs e)
{
    string selectedOption = comboBox.SelectedItem as ComboBoxItem;
    MessageBox.Show($"You selected: {selectedOption.Content}");
}

6. Aplikacja z Timerem (Odliczanie czasu)

Aplikacja z Timerem, która odlicza czas i wyświetla upływające sekundy.

<Window x:Class="TimerExample.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        Title="Timer Example" Height="200" Width="300">
    <Grid>
        <TextBlock Name="textBlock" HorizontalAlignment="Center" VerticalAlignment="Center" FontSize="30"/>
    </Grid>
</Window>

W C#: ---------------------------------------------------------------------------------------------------------------------------------------------------

private DispatcherTimer timer;
private int seconds = 0;

public MainWindow()
{
    InitializeComponent();
    timer = new DispatcherTimer();
    timer.Interval = TimeSpan.FromSeconds(1);
    timer.Tick += Timer_Tick;
    timer.Start();
}

private void Timer_Tick(object sender, EventArgs e)
{
    seconds++;
    textBlock.Text = $"Time: {seconds} sec";
}
