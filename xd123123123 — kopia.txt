1. Galeria zdjęć z filtrem po kategoriach

XAML (MainWindow.xaml):

<Window x:Class="PhotoGallery.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        Title="Galeria zdjęć" Height="450" Width="800">
    <Grid>
        <ComboBox Name="categoryComboBox" VerticalAlignment="Top" Margin="10,10,10,0" SelectionChanged="CategoryComboBox_SelectionChanged">
            <ComboBoxItem Content="Krajobrazy"/>
            <ComboBoxItem Content="Zwierzęta"/>
            <ComboBoxItem Content="Portrety"/>
        </ComboBox>

        <ListView Name="photoListView" Margin="10,50,10,10">
            <ListView.View>
                <GridView>
                    <GridViewColumn Header="Zdjęcie" DisplayMemberBinding="{Binding Path=ImagePath}" Width="100"/>
                    <GridViewColumn Header="Opis" DisplayMemberBinding="{Binding Path=Description}" Width="300"/>
                </GridView>
            </ListView.View>
        </ListView>
    </Grid>
</Window>

C# (MainWindow.xaml.cs):

using System.Collections.Generic;
using System.Windows;

namespace PhotoGallery
{
    public partial class MainWindow : Window
    {
        public MainWindow()
        {
            InitializeComponent();
        }

        private void CategoryComboBox_SelectionChanged(object sender, System.Windows.Controls.SelectionChangedEventArgs e)
        {
            string selectedCategory = (categoryComboBox.SelectedItem as System.Windows.Controls.ComboBoxItem).Content.ToString();
            LoadPhotosByCategory(selectedCategory);
        }

        private void LoadPhotosByCategory(string category)
        {
            List<Photo> photos = new List<Photo>();

            // Przykładowe zdjęcia, można zastąpić prawdziwymi ścieżkami.
            if (category == "Krajobrazy")
            {
                photos.Add(new Photo { ImagePath = "landscape1.jpg", Description = "Krajobraz 1" });
                photos.Add(new Photo { ImagePath = "landscape2.jpg", Description = "Krajobraz 2" });
            }
            else if (category == "Zwierzęta")
            {
                photos.Add(new Photo { ImagePath = "animal1.jpg", Description = "Zwierzę 1" });
                photos.Add(new Photo { ImagePath = "animal2.jpg", Description = "Zwierzę 2" });
            }
            else if (category == "Portrety")
            {
                photos.Add(new Photo { ImagePath = "portrait1.jpg", Description = "Portret 1" });
                photos.Add(new Photo { ImagePath = "portrait2.jpg", Description = "Portret 2" });
            }

            photoListView.ItemsSource = photos;
        }
    }

    public class Photo
    {
        public string ImagePath { get; set; }
        public string Description { get; set; }
    }
}

3. Slajd pokaz z automatycznym przewijaniem

XAML (MainWindow.xaml):

<Window x:Class="SlideshowApp.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        Title="Slideshow" Height="450" Width="800">
    <Grid>
        <Image Name="slideshowImage" HorizontalAlignment="Center" VerticalAlignment="Center"/>
        <Button Content="Start" VerticalAlignment="Bottom" HorizontalAlignment="Center" Margin="0,0,0,50" Click="StartButton_Click"/>
    </Grid>
</Window>

C# (MainWindow.xaml.cs):

using System;
using System.Windows;
using System.Windows.Threading;

namespace SlideshowApp
{
    public partial class MainWindow : Window
    {
        private string[] imagePaths = { "image1.jpg", "image2.jpg", "image3.jpg" };
        private int currentIndex = 0;
        private DispatcherTimer timer;

        public MainWindow()
        {
            InitializeComponent();
            timer = new DispatcherTimer();
            timer.Interval = TimeSpan.FromSeconds(2); // Zmienia co 2 sekundy
            timer.Tick += Timer_Tick;
        }

        private void StartButton_Click(object sender, RoutedEventArgs e)
        {
            timer.Start();
        }

        private void Timer_Tick(object sender, EventArgs e)
        {
            slideshowImage.Source = new System.Windows.Media.Imaging.BitmapImage(new Uri(imagePaths[currentIndex], UriKind.Relative));
            currentIndex = (currentIndex + 1) % imagePaths.Length; // Przechodzi do następnego zdjęcia, po ostatnim zaczyna od początku
        }
    }
}
